import java.util.*;

public class main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a[]=new int[5];
        a[0]=10;
        a[1]=20;
        a[2]=30;
        a[3]=40;
        int n=4;
        System.out.println("enter value:");
        int value=sc.nextInt();
        System.out.println("enter index(0 to"+n+"):");
        int pos=sc.nextInt();
        if(pos<0 || pos>n||n==a.length){
            System.out.println("invvalid insertion");
        }else{
            for(int i=n;i>pos;i--){
                a[i]=a[i-1];
        }
            a[pos]=value;
            n++;
            for(int i=0;i<n;i++){
                System.out.print(a[i]+" ");
            }
        }
        sc.close();
    }
}

import java.util.*;

public class main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a[]={10,20,30,50,40,60};
        int key=40;
        int low=0;
        int high=a.length-1;
        int pos=-1;
        while(low<=high){
            int mid=(low+high)/2;
            if (a[mid]==key){
                pos=mid;
                break;
            }else if (key>a[mid]){
                low=mid+1;
            }else{
                high=mid-1;
            }
        }
        if (pos!=-1)
            System.out.println("found at index"+pos);
        else
            System.out.println("not found");
    }
}
